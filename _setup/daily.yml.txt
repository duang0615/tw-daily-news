name: 每日更新新聞

on:
  schedule:
    # UTC 00:00 = 台灣時間 08:00
    - cron: "0 0 * * *"
  workflow_dispatch:        # 也可以在 Actions 頁面手動按一下就跑

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: 抓新聞、產出頁面
        run: python build_news.py --no-hub

      - name: 有變動才 commit
        run: |
          git config user.name  "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add index.html news.json archive/
          if git diff --staged --quiet; then
            echo "沒有變動，跳過"
          else
            git commit -m "每日更新 $(TZ=Asia/Taipei date '+%Y-%m-%d %H:%M')"
            git push
          fi
